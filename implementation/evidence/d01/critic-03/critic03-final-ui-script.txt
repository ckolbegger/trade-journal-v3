import {webkit,assert,contexts,results,events,write,pause,record,launch,ready,shot,state,same,installFaults,fault,fill,scan} from './critic03-helpers.mjs';
import {key,move,clickKey,typeKey} from './critic03-key-helpers.mjs';
function ratio(a,b){const v=s=>s.match(/[\d.]+/g).slice(0,3).map(Number).map(x=>x/255).map(x=>x<=.04045?x/12.92:((x+.055)/1.055)**2.4);const l=s=>v(s).reduce((z,x,i)=>z+x*[.2126,.7152,.0722][i],0);let x=l(a),y=l(b);return(Math.max(x,y)+.05)/(Math.min(x,y)+.05)}
async function run(){for(const width of [390,1440]){const x=await launch(`final-ui-${width}`,webkit,width,`/private/tmp/tjv3-critic03-keyboard-${width}`);await installFaults(x.c);const p=x.p;await ready(p);await clickKey(p,p.getByRole('button',{name:'New Plan',exact:true}));for(const [label,value]of [['Stock symbol','STALE'+width],['Quantity (shares)','100'],['Intended entry (USD)','50'],['Stop (underlying price, USD)','45'],['Target (underlying price, USD)','60'],['Why this trade, why now?','Stale keyboard thesis'],['What would invalidate the thesis?','Stale keyboard invalidation']])await typeKey(p,p.getByLabel(label,{exact:true}),value);
 const tab=await x.c.newPage();await ready(tab);await tab.getByRole('button',{name:'Workspace',exact:true}).click();await tab.getByLabel('Institution name',{exact:true}).fill('Competing catalog '+width);await tab.getByRole('button',{name:'Save Institution',exact:true}).click();await pause(200);const b=await state(p);await clickKey(p,p.getByRole('button',{name:'Confirm Plan',exact:true}));await p.getByText('Brokerage records changed in another tab. Review your selections before confirming again.',{exact:true}).waitFor();await same(p,b,`keyboard-stale-${width}-no-write`);assert.equal(await p.getByLabel('Stock symbol',{exact:true}).inputValue(),'STALE'+width);await scan(p,`keyboard-stale-${width}`);await clickKey(p,p.getByRole('button',{name:'Review current references',exact:true}));await tab.close();
 await fault(p,{abortKind:'Plan'});await clickKey(p,p.getByRole('button',{name:'Confirm Plan',exact:true}));await p.getByText('Saving failed. Nothing was recorded. Your entered values are preserved. Retry startup before saving again.',{exact:true}).waitFor();await same(p,b,`keyboard-Plan-abort-${width}-no-write`);await scan(p,`keyboard-Plan-abort-${width}`);await fault(p,{});await clickKey(p,p.getByRole('button',{name:'Retry startup',exact:true}));await p.getByRole('button',{name:'Confirm Plan',exact:true}).waitFor();
 // Supplemental CSS 200% stress reflow. This is explicitly not native browser-chrome zoom evidence.
 await p.evaluate(()=>document.documentElement.style.zoom='2');const css=await p.evaluate(()=>({method:'CSS zoom supplemental only',zoom:getComputedStyle(document.documentElement).zoom,innerWidth,clientWidth:document.documentElement.clientWidth,scrollWidth:document.documentElement.scrollWidth}));await shot(p,`supplemental-css200-${width}`);record(`supplemental-reflow-${width}`,css);await p.evaluate(()=>document.documentElement.style.zoom='');
 await move(p,p.getByLabel('Stock symbol',{exact:true}));const colors=await p.getByLabel('Stock symbol',{exact:true}).evaluate(e=>{const s=getComputedStyle(e),body=getComputedStyle(document.body);return {text:s.color,background:s.backgroundColor,border:s.borderColor,focus:s.outlineColor,focusStyle:s.outlineStyle,focusWidth:s.outlineWidth,page:body.backgroundColor}});record(`contrast-${width}`,{colors,text:ratio(colors.text,colors.background),essentialBorder:ratio(colors.border,colors.background),focusOnPage:ratio(colors.focus,colors.page)});assert(ratio(colors.text,colors.background)>=4.5);assert(ratio(colors.border,colors.background)>=3);assert(ratio(colors.focus,colors.page)>=3);await shot(p,`focus-${width}`);
 await clickKey(p,p.getByRole('button',{name:'Cancel',exact:true}));await same(p,b,`final-cancel-${width}-no-write`);await x.c.close();}await write('final-ui-results.json',results);await write('final-ui-console.json',events)}
run().catch(async e=>{console.error(e);await write('final-ui-error.txt',e.stack);process.exitCode=1}).finally(async()=>{for(const c of contexts)await c.close().catch(()=>{})});
