import {webkit,chromium,firefox,assert,contexts,results,events,write,pause,record,launch,ready,shot,state,same,installFaults,fault,setup,fill,scan,inventory,corrupt,restore} from './critic03-helpers.mjs';
async function main(){
 const x=await launch('readiness');await installFaults(x.c);const p=x.p;await ready(p);await setup(p,'recovery');await p.getByRole('button',{name:'New Plan',exact:true}).click();await fill(p,'READ');await p.getByRole('button',{name:'Confirm Plan',exact:true}).click();await p.getByRole('heading',{name:'READ · Planned',exact:true}).waitFor();await p.getByRole('button',{name:'Trades',exact:true}).click();const b=await state(p);await inventory(p,'readiness-native');
 for(const check of ['transaction','capacity','control','inventory','simultaneous']){
  let asset;const f={};if(check==='inventory'||check==='simultaneous')asset=await corrupt(p);if(check==='simultaneous'){f.transaction=true;f.capacity=true;f.control=true}else if(check!=='inventory')f[check]=true;
  await fault(p,f);await p.getByRole('button',{name:'Workspace',exact:true}).click(); // preserve ordinary route while failure starts
  // Retry startup becomes available after a native late mutation gate catches the fault.
  await p.getByLabel('Institution name',{exact:true}).fill(`blocked ${check}`);await p.getByRole('button',{name:'Save Institution',exact:true}).click();await p.getByRole('heading',{name:'Recovery Only',exact:true}).waitFor();await same(p,b,`existing-${check}-unchanged`);
  const reasons=await p.locator('[data-issue]').evaluateAll(es=>es.map(e=>e.getAttribute('data-issue')));if(check==='simultaneous')for(const k of ['transaction','capacity','control','inventory'])assert(reasons.includes(k),`${check} lacks ${k}`);else assert(reasons.includes(check));assert(await p.getByRole('button',{name:'New Plan',exact:true}).isDisabled());await scan(p,`recovery-${check}`);
  await p.getByRole('button',{name:'Trades',exact:true}).click();await p.getByRole('button',{name:/READ · Planned/}).click();await p.getByText('View history',{exact:true}).click();await shot(p,`recovery-${check}-history`);await same(p,b,`existing-${check}-history-no-write`);
  record(`existing-${check}`,{reasons,body:await p.locator('body').innerText(),trace:await p.evaluate(()=>window.__criticTrace)});
  if(asset)await restore(p,asset);await fault(p,{});await p.getByRole('button',{name:'Retry startup',exact:true}).click();await p.getByRole('button',{name:'New Plan',exact:true}).waitFor();assert(await p.getByRole('heading',{name:'Recovery Only',exact:true}).count()===0);
 }
 // Known root read fails at actual IndexedDB object-store read boundary. No bytes replaced.
 await fault(p,{unreadable:true});await p.reload(); // init reset; set fault while new page startup held is instead done with a persistent context init script below
 await p.getByRole('button',{name:'Workspace',exact:true}).waitFor();await fault(p,{unreadable:true});await p.getByRole('button',{name:'Workspace',exact:true}).click();await p.getByLabel('Institution name',{exact:true}).fill('unreadable-blocked');await p.getByRole('button',{name:'Save Institution',exact:true}).click();await p.getByRole('heading',{name:'IntegrityBlocked',exact:true}).waitFor();await scan(p,'unreadable-root');await fault(p,{});await same(p,b,'unreadable-root-preserved');await p.getByRole('button',{name:'Retry startup',exact:true}).click();await p.getByRole('button',{name:'Workspace',exact:true}).waitFor();
 // Fresh proved-empty profiles independently fail each actual host check; cached corruption is introduced only after genuine first successful release staging, before Workspace creation.
 for(const check of ['transaction','capacity','control','inventory','simultaneous']){
  const f=await launch(`fresh-${check}`);await installFaults(f.c);await ready(f.p);const fresh=await state(f.p);assert.equal(fresh.roots.length,0);let asset;const flags={};if(check==='inventory'||check==='simultaneous')asset=await corrupt(f.p);if(check==='simultaneous'){flags.transaction=true;flags.capacity=true;flags.control=true}else if(check!=='inventory')flags[check]=true;await fault(f.p,flags);await f.p.getByLabel('Workspace time zone',{exact:true}).fill('America/New_York');await f.p.getByRole('button',{name:'Create Workspace',exact:true}).click();await f.p.getByRole('heading',{name:'Unsupported',exact:true}).waitFor();await same(f.p,fresh,`fresh-${check}-no-initialization`);const reasons=await f.p.locator('[data-issue]').evaluateAll(es=>es.map(e=>e.getAttribute('data-issue')));if(check==='simultaneous')for(const k of ['transaction','capacity','control','inventory'])assert(reasons.includes(k));else assert(reasons.includes(check));await scan(f.p,`unsupported-${check}`);record(`fresh-${check}`,{reasons,trace:await f.p.evaluate(()=>window.__criticTrace)});if(asset)await restore(f.p,asset);await fault(f.p,{});await f.p.getByRole('button',{name:'Retry startup',exact:true}).click();await f.p.getByLabel('Workspace time zone',{exact:true}).waitFor();await same(f.p,fresh,`fresh-${check}-retry-no-data`);await f.c.close();
 }
 // Actual native other hosts: retain reported quotas; do not change budgets or quota to obtain Writable.
 for(const [name,kind]of [['chromium',chromium],['firefox',firefox]]){const h=await launch(`host-${name}`,kind);await ready(h.p);record(`host-${name}`,{body:await h.p.locator('body').innerText(),native:await h.p.evaluate(async()=>({ua:navigator.userAgent,secure:isSecureContext,quota:await navigator.storage.estimate(),persisted:await navigator.storage.persisted(),controller:navigator.serviceWorker.controller?.scriptURL})),state:await state(h.p)});await shot(h.p,`host-${name}`);await h.c.close()}
 await write('readiness-results.json',results);await write('readiness-console.json',events);
}
main().catch(async e=>{console.error(e);await write('readiness-error.txt',e.stack);process.exitCode=1}).finally(async()=>{for(const c of contexts)await c.close().catch(()=>{})});
