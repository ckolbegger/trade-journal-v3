import {launch,ready,plan,confirm,state,shot,log,assert,fs,out} from './critic02-common.mjs';

 await plan(p,'TRACE2');await p.evaluate(()=>window.__txTrace=[]);await confirm(p);const trace=await p.evaluate(()=>window.__txTrace);const lastMutation=trace.findLastIndex(x=>x.event==='complete'&&x.mode==='readwrite');assert.ok(lastMutation>=0);assert.equal(trace.slice(lastMutation+1).some(x=>x.event==='begin'),false);await log('one-call-native-transaction-trace',{trace,noReadAfterCommit:true,rendered:await p.locator('main').innerText()});await p.getByRole('button',{name:'Trades',exact:true}).click();const before=await state(p);const row=before.lifecycle.find(x=>x.symbol==='TRACE2');await p.evaluate(async row=>{const db=await new Promise(res=>{const r=indexedDB.open('trade-journal-v3');r.onsuccess=()=>res(r.result)});const tx=db.transaction('lifecycle','readwrite');tx.objectStore('lifecycle').put({...row,lifecycle:'Open'});await new Promise((res,rej)=>{tx.oncomplete=res;tx.onerror=()=>rej(tx.error)});db.close()},row);const corrupt=await state(p);await p.getByRole('button',{name:/TRACE2 · Planned/}).click();await p.getByRole('alert').getByText('IntegrityBlocked',{exact:true}).waitFor();assert.deepEqual(await state(p),corrupt);await shot(p,'b02-lifecycle-integrity');await p.evaluate(async row=>{const db=await new Promise(res=>{const r=indexedDB.open('trade-journal-v3');r.onsuccess=()=>res(r.result)});const tx=db.transaction('lifecycle','readwrite');tx.objectStore('lifecycle').put(row);await new Promise(res=>tx.oncomplete=res);db.close()},row);assert.deepEqual(await state(p),before);await log('b02-integrity',{controlledAgreementRowChanged:'lifecycle TRACE Planned to Open',normalDetailBlocked:true,noReadRepair:true,exactOriginalRestored:true});await p.getByRole('button',{name:'New Plan',exact:true}).click();await fs.writeFile(out+'/b05-form-tree.txt',await p.locator('body').ariaSnapshot());
}catch(e){await shot(p,'extra-failure');await log('extra-error',{error:String(e),stack:e.stack});throw e}finally{await c.close()}
