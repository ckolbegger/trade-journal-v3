import {launch,ready,onboard,plan,confirm,state,shot,log,assert,url,chromium,firefox} from './critic02-common.mjs';
const results=[];
const backups=new Map();
async function fault(p,f){
 if(backups.has(p)){await p.evaluate(async b=>{const cache=await caches.open(b.name);await cache.put('/icon-192.png',new Response(new Uint8Array(b.bytes),{headers:{'Content-Type':'image/png'}}))},backups.get(p));backups.delete(p)}
 if(f.inventory){const b=await p.evaluate(async()=>{const name=(await caches.keys()).find(n=>n.startsWith('trade-journal-release-'));const cache=await caches.open(name);const original=await cache.match('/icon-192.png');const bytes=Array.from(new Uint8Array(await original.arrayBuffer()));await cache.put('/icon-192.png',new Response('critic-02 actual damaged cached icon'));return {name,bytes}});backups.set(p,b)}
 await p.evaluate(f=>{sessionStorage.setItem('critic02-faults',JSON.stringify(f));window.__faults=f},f);
}
for(const kind of ['transaction','capacity','control','inventory','simultaneous']){
 const c=await launch('fresh-'+kind);const p=c.pages()[0];try{await ready(p);await fault(p,kind==='simultaneous'?{transaction:true,capacity:true}:{[kind]:true});await p.reload();await p.getByRole('heading',{name:'Unsupported',exact:true}).waitFor({timeout:20000});const issues=await p.locator('[data-issue]').evaluateAll(es=>es.map(e=>e.getAttribute('data-issue')));const expected=kind==='simultaneous'?['transaction','capacity']:[kind];for(const k of expected)assert.ok(issues.includes(k));const bytes=await state(p);assert.equal(bytes.roots.length,0);assert.equal(bytes.versions.length,0);assert.equal(bytes.diagnostic.length,0);await shot(p,'b03-fresh-'+kind);results.push({case:'fresh-'+kind,issues,unchangedNoWorkspace:true});}finally{await c.close()}
}
const c=await launch('main');const p=c.pages()[0];try{await ready(p);const baseline=await state(p);for(const kind of ['transaction','capacity','control','inventory','simultaneous']){await fault(p,kind==='simultaneous'?{transaction:true,capacity:true}:{[kind]:true});await p.reload();await p.getByRole('heading',{name:'Recovery Only',exact:true}).waitFor({timeout:20000});await p.getByRole('button',{name:/AAPL · Planned/}).click();await p.getByText('View history',{exact:true}).click();await p.getByText('Plan fact:',{exact:false}).waitFor();assert.ok(await p.getByRole('button',{name:'New Plan',exact:true}).isDisabled());await p.getByRole('button',{name:'Workspace',exact:true}).click();await p.getByText('Reference changes are disabled in Recovery Only.',{exact:true}).waitFor();assert.deepEqual(await state(p),baseline);await shot(p,'b03-recovery-'+kind);results.push({case:'recovery-'+kind,readDetailHistory:true,mutationsDisabled:true,unchanged:true});await fault(p,{});await p.getByRole('button',{name:'Retry startup',exact:true}).click();await p.getByRole('heading',{name:'Recovery Only',exact:true}).waitFor({state:'hidden'});}
 await fault(p,{unreadable:true});await p.reload();await p.getByRole('heading',{name:'IntegrityBlocked',exact:true}).waitFor();assert.equal(await p.getByRole('button',{name:'Create Workspace',exact:true}).count(),0);assert.deepEqual(await state(p),baseline);await shot(p,'b03-unreadable-known-root');results.push({case:'unreadable-known-root',noInitialization:true,preservedRootAndHistories:true});await fault(p,{});await p.getByRole('button',{name:'Retry startup',exact:true}).click();await p.getByRole('button',{name:'New Plan',exact:true}).waitFor();
 await plan(p,'LATEFAIL');await fault(p,{capacity:true});await p.getByRole('button',{name:'Confirm Plan',exact:true}).click();await p.getByRole('heading',{name:'Recovery Only',exact:true}).waitFor();assert.equal(await p.getByLabel('Stock symbol',{exact:true}).inputValue(),'LATEFAIL');assert.deepEqual(await state(p),baseline);await shot(p,'b03-late-capacity-failure');results.push({case:'late-failure',inputPreserved:true,rejectedBeforeMutation:true,unchanged:true});
 await log('b03-readiness',results);
}finally{await c.close()}
for(const [name,type] of [['chromium',chromium],['firefox',firefox]]){let cc;try{cc=await launch('host-'+name,1440,type);const pp=cc.pages()[0];await ready(pp);await pp.getByRole('heading',{name:'Unsupported',exact:true}).waitFor();await log('host-'+name,{ua:await pp.evaluate(()=>navigator.userAgent),secure:await pp.evaluate(()=>isSecureContext),estimate:await pp.evaluate(()=>navigator.storage.estimate()),body:await pp.locator('body').innerText(),state:await state(pp)});await shot(pp,'host-'+name)}catch(e){await log('host-'+name+'-error',{error:String(e)})}finally{await cc?.close()}}
