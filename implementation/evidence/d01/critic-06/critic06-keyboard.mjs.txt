import {chromium,assert,write,logs,watch,shot,audit,state,ready,fields} from './critic06-lib.mjs';
let browser;
try{
browser=await chromium.launch({channel:'chrome',headless:true});let trace=[];
for(let width of [390,1440]){let ctx=await browser.newContext({viewport:{width,height:1000}}),p=await ctx.newPage();watch(p);await ready(p);
async function tabTo(locator){for(let i=0;i<60;i++){if(await locator.evaluate(el=>el===document.activeElement))return;await p.keyboard.press('Tab');trace.push({width,...await p.evaluate(()=>({tag:document.activeElement?.tagName,text:document.activeElement?.textContent?.slice(0,90),name:document.activeElement?.getAttribute('name')}))})}throw new Error('Tab target not reachable')}
async function type(label,value){await tabTo(p.getByLabel(label,{exact:true}));await p.keyboard.type(value)}
async function enter(name){await tabTo(p.getByRole('button',{name,exact:true}));await p.keyboard.press('Enter')}
await enter('Create Workspace');await p.getByRole('heading',{name:'Choose your time zone',exact:true}).waitFor();assert.equal((await state(p)).roots.length,0);await shot(p,'b05-tab-zone-error-'+width);await type('Workspace time zone','America/New_York');await enter('Create Workspace');await p.getByRole('heading',{name:'Add your brokerage records',exact:true}).waitFor();await type('Institution name','Keyboard '+width);await enter('Save Institution');await type('Account name','Taxable '+width);await enter('Save Account');await p.getByRole('heading',{name:'Planned Trades',exact:true}).waitFor();await enter('New Plan');let before=await state(p);await enter('Confirm Plan');await p.getByRole('alert').waitFor();assert.deepEqual(await state(p),before);for(let [label,value] of fields)await type(label,label==='Stock symbol'?'TAB'+width:value);await tabTo(p.getByRole('radio',{name:'Unanswered',exact:true}));await p.keyboard.press('ArrowRight');await enter('Confirm Plan');await p.getByRole('heading',{name:'TAB'+width+' · Planned',exact:true}).waitFor();await shot(p,'b05-tab-complete-'+width);await audit(p,'b05-tab-complete-'+width);await tabTo(p.getByText('View history',{exact:true}));await p.keyboard.press('Enter');await shot(p,'b05-tab-history-'+width);await write('b05-tab-state-'+width+'.json',await state(p));await ctx.close()}
await write('b05-tab-trace.json',trace);await write('keyboard-complete.json',{status:'passed'});
}catch(e){await write('keyboard-failure.txt',e.stack||String(e));throw e}finally{if(browser)await browser.close();await write('keyboard-console.json',logs)}
